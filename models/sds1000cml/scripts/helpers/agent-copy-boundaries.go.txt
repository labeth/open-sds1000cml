// Temporary-file characterization; no command or device is executed.
func TestModelReviewAgentCopyMode(t *testing.T) {
	root := t.TempDir()
	src, dst := root+"/source", root+"/target"
	if err := os.WriteFile(src, []byte("payload"), 0o600); err != nil { t.Fatal(err) }
	if err := os.WriteFile(dst+".tmp", []byte("old"), 0o600); err != nil { t.Fatal(err) }
	if err := os.Chmod(dst+".tmp", 0o600); err != nil { t.Fatal(err) }
	if _, err := copyExecFile(src, dst); err != nil { t.Fatal(err) }
	info, err := os.Stat(dst)
	if err != nil { t.Fatal(err) }
	if info.Mode().Perm() != 0o600 { t.Fatalf("expected retained temporary mode 0600, got %o", info.Mode().Perm()) }
	data, err := os.ReadFile(dst)
	if err != nil || string(data) != "payload" { t.Fatalf("copy content %q: %v", data, err) }
	t.Log("copyExecFile preserves existing .tmp mode 0600 despite its 0755 comment")
}
