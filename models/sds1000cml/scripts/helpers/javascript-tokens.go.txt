package main

import (
 "crypto/sha256"
 "encoding/hex"
 "encoding/json"
 "os"
 sitter "github.com/tree-sitter/go-tree-sitter"
 javascript "github.com/tree-sitter/tree-sitter-javascript/bindings/go"
)

func main() {
 src, err := os.ReadFile(os.Args[1]); if err != nil { panic(err) }
 parser := sitter.NewParser(); defer parser.Close()
 if err := parser.SetLanguage(sitter.NewLanguage(javascript.Language())); err != nil { panic(err) }
 tree := parser.Parse(src, nil); defer tree.Close()
 if tree.RootNode().HasError() { panic("JavaScript syntax error") }
 tokens := [][2]string{}
 var visit func(*sitter.Node)
 visit = func(n *sitter.Node) {
  if n.Kind() == "comment" { return }
  if n.ChildCount() == 0 { tokens = append(tokens, [2]string{n.Kind(), n.Utf8Text(src)}); return }
  for i := uint(0); i < n.ChildCount(); i++ { visit(n.Child(i)) }
 }
 visit(tree.RootNode())
 data, err := json.Marshal(tokens); if err != nil { panic(err) }
 hash := sha256.Sum256(data)
 json.NewEncoder(os.Stdout).Encode(map[string]any{"tokenSHA256": hex.EncodeToString(hash[:]), "tokens": len(tokens)})
}
